import sqlite3

from pathlib import Path
from flask import current_app, g


# ---------------------------------------------------------
# DATABASE PATH RESOLUTION
# ---------------------------------------------------------

def get_database_path():
    """
    Returns the absolute path of the SQLite database.

    Relative paths are resolved from the project root.
    """

    database_path = Path(
        current_app.config["DATABASE_PATH"]
    )

    if not database_path.is_absolute():
        project_root = Path(__file__).resolve().parents[2]
        database_path = project_root / database_path

    return database_path.resolve()


# ---------------------------------------------------------
# DATABASE CONNECTION
# ---------------------------------------------------------

def get_db():
    """
    Returns the current request's SQLite connection.

    A single connection is reused during the current
    Flask application context.
    """

    if "db" not in g:

        database_path = get_database_path()

        database_path.parent.mkdir(
            parents=True,
            exist_ok=True
        )

        connection = sqlite3.connect(
            database_path,
            detect_types=sqlite3.PARSE_DECLTYPES
        )

        # Return database rows as dictionary-like objects.
        connection.row_factory = sqlite3.Row

        # Enforce foreign key constraints.
        connection.execute("PRAGMA foreign_keys = ON")

        # Configure SQLite busy timeout.
        connection.execute("PRAGMA busy_timeout = 5000")

        g.db = connection

    return g.db


# ---------------------------------------------------------
# CLOSE DATABASE CONNECTION
# ---------------------------------------------------------

def close_db(error=None):
    """
    Closes the database connection at the end of
    the Flask application context.
    """

    database = g.pop("db", None)

    if database is not None:
        database.close()


# ---------------------------------------------------------
# DATABASE INITIALIZATION HOOK
# ---------------------------------------------------------

def init_app(app):
    """
    Registers database cleanup with the Flask application.
    """

    app.teardown_appcontext(close_db)


# ---------------------------------------------------------
# DATABASE TRANSACTION HELPER
# ---------------------------------------------------------

def execute_transaction(callback):
    """
    Executes a group of database operations inside
    a transaction.

    Parameters:
        callback: Function receiving the database connection.

    Returns:
        Result returned by the callback.

    Rolls back changes if an exception occurs.
    """

    database = get_db()

    try:
        with database:
            result = callback(database)

        return result

    except Exception:
        raise

import sqlite3

from pathlib import Path

from app.database.connection import get_db, get_database_path


# ---------------------------------------------------------
# SCHEMA FILE LOCATION
# ---------------------------------------------------------

SCHEMA_PATH = (
    Path(__file__).resolve().parent / "schema.sql"
)


# ---------------------------------------------------------
# INITIALIZE DATABASE
# ---------------------------------------------------------

def initialize_database(app):
    """
    Initializes the StrategySim SQLite database.

    This function:
        1. Creates the database directory.
        2. Reads the SQL schema.
        3. Creates all required tables and indexes.
        4. Checks database integrity.
        5. Checks foreign key consistency.

    Existing data is preserved because the schema uses
    CREATE TABLE IF NOT EXISTS.
    """

    print("\nInitializing StrategySim database...")

    # -----------------------------------------------------
    # CHECK SCHEMA FILE
    # -----------------------------------------------------

    if not SCHEMA_PATH.exists():

        raise FileNotFoundError(
            f"Database schema not found: {SCHEMA_PATH}"
        )

    schema_sql = SCHEMA_PATH.read_text(
        encoding="utf-8"
    )

    # -----------------------------------------------------
    # ACTIVATE FLASK APPLICATION CONTEXT
    # -----------------------------------------------------

    with app.app_context():

        # Resolve the database location only after the
        # application context has been activated.

        database_path = get_database_path()

        database_path.parent.mkdir(
            parents=True,
            exist_ok=True
        )

        # -------------------------------------------------
        # CREATE TABLES AND INDEXES
        # -------------------------------------------------

        database = get_db()

        try:

            database.executescript(schema_sql)

            database.commit()

        except sqlite3.Error as error:

            database.rollback()

            raise RuntimeError(
                f"Database schema initialization failed: {error}"
            ) from error

        # -------------------------------------------------
        # DATABASE INTEGRITY CHECK
        # -------------------------------------------------

        integrity_result = database.execute(
            "PRAGMA integrity_check"
        ).fetchone()

        if integrity_result[0] != "ok":

            raise RuntimeError(
                "SQLite database integrity check failed: "
                f"{integrity_result[0]}"
            )

        # -------------------------------------------------
        # FOREIGN KEY CHECK
        # -------------------------------------------------

        foreign_key_errors = database.execute(
            "PRAGMA foreign_key_check"
        ).fetchall()

        if foreign_key_errors:

            raise RuntimeError(
                "Database foreign key validation failed: "
                f"{len(foreign_key_errors)} issue(s) found."
            )

        # -------------------------------------------------
        # COUNT CREATED TABLES
        # -------------------------------------------------

        tables = database.execute(
            """
            SELECT name
            FROM sqlite_master
            WHERE type = 'table'
              AND name NOT LIKE 'sqlite_%'
            ORDER BY name
            """
        ).fetchall()

        table_count = len(tables)

    # -----------------------------------------------------
    # INITIALIZATION SUCCESS
    # -----------------------------------------------------

    print("-" * 55)

    print(
        "StrategySim database initialized successfully."
    )

    print(
        f"Database location: {database_path}"
    )

    print(
        f"Tables created/verified: {table_count}"
    )

    print(
        "Integrity check: PASSED"
    )

    print(
        "Foreign key check: PASSED"
    )

    print("-" * 55)

    return database_path

import os

    current_user
)

from flask_wtf.csrf import CSRFProtect

from app.config import configurations

    init_app as init_database_app,
    initialize_database
)


# ---------------------------------------------------------
# FLASK EXTENSIONS
# ---------------------------------------------------------

login_manager = LoginManager()
csrf = CSRFProtect()


# ---------------------------------------------------------
# APPLICATION FACTORY
# ---------------------------------------------------------

def create_app(config_name=None):
    """
    Creates and configures the StrategySim Flask application.
    """

    app = Flask(
        __name__,
        instance_relative_config=True
    )

    # -----------------------------------------------------
    # LOAD CONFIGURATION
    # -----------------------------------------------------

    if config_name is None:
        config_name = os.getenv(
            "FLASK_ENV",
            "development"
        )

    config_class = configurations.get(
        config_name,
        configurations["development"]
    )

    app.config.from_object(config_class)

    # -----------------------------------------------------
    # VALIDATE SECRET KEY
    # -----------------------------------------------------

    if not app.config.get("SECRET_KEY"):
        raise RuntimeError(
            "SECRET_KEY is missing. "
            "Generate a secure secret key and configure it "
            "in the .env file."
        )

    # -----------------------------------------------------
    # CREATE INSTANCE DIRECTORY
    # -----------------------------------------------------

    os.makedirs(
        app.instance_path,
        exist_ok=True
    )

    # -----------------------------------------------------
    # INITIALIZE FLASK EXTENSIONS
    # -----------------------------------------------------

    login_manager.init_app(app)

    login_manager.login_view = "auth.login"

    login_manager.login_message = (
        "Please sign in to access this page."
    )

    login_manager.login_message_category = "warning"

    login_manager.session_protection = "strong"

    csrf.init_app(app)

    # -----------------------------------------------------
    # INITIALIZE DATABASE
    # -----------------------------------------------------

    init_database_app(app)

    initialize_database(app)

    # -----------------------------------------------------

@login_manager.user_loader
def load_user(user_id):
    """
    Retrieves an active authenticated user from SQLite.

    Inactive accounts are not restored into authenticated
    sessions.
    """

    from app.users.models import User

    user = User.get_by_id(user_id)

    if user is None or not user.is_active:
        return None

    return user
